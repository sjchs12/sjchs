    'Arad': 366, 'Zerind': 374, 'Timisoara': 329,

def a_star_search(graph, heuristic, start, goal):
    open_list = []
    heapq.heappush(open_list, (heuristic[start], 0, start, [start]))
    best_cost = {start: 0} 
 
    while open_list: 
        f, g, city, path = heapq.heappop(open_list) 
 
        if g != best_cost.get(city): 
            continue
        
        if city == goal:
            return path, g
        
        for neighbor, distance in graph[city].items():
            new_cost = g + distance 
            
            if new_cost < best_cost.get(neighbor, float('inf')):
                best_cost[neighbor] = new_cost
                new_f = new_cost + heuristic[neighbor]
                
                heapq.heappush(
                    open_list,(new_f, new_cost, neighbor, path + [neighbor]) 
                ) 
 
    return None, float('inf') 
 
 
# Example usage start_city = 'Arad' goal_city = 'Bucharest' 
 
path, cost = a_star_search( 
    romania_map, heuristic, start_city, goal_city 
) 
 
if path: 
    print("Path found:", " -> ".join(path))
    print("Total distance:", cost, "km")
else: 
    print("No path found.")
