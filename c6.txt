Aim - Digital Signature Using RSA 
 
Title - To implement an RSA-based signature algorithm in Python and verify the integrity and authenticity of a digitally signed message. 
 
Terminal - pip install cryptography 
 
 
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes
from cryptography.exceptions import InvalidSignature 
 
# Generate RSA private key
private_key = rsa.generate_private_key( 
    public_exponent=65537,
    key_size=2048 
) 
 
# Generate public key
public_key = private_key.public_key() 
 
# Create message
message = "This is a secure message."
data = message.encode() 
 
# Create digital signature
signature = private_key.sign(
    data,
    padding.PSS(
        mgf=padding.MGF1(hashes.SHA256()),
        salt_length=padding.PSS.MAX_LENGTH 
    ), 
    hashes.SHA256() 
) 
 
print("Original message:")
print(message) 
 
print("\nDigital Signature:")
print(signature.hex()) 
 
# Verify signature 
try: 
    public_key.verify(
        signature,
        data,
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256())
            salt_length=padding.PSS.MAX_LENGTH 
        ), 
        hashes.SHA256() 
    ) 
    
    print("\nSignature Verification: SUCCESS")
    print("The message is authentic and unchanged.") 

except InvalidSignature:
    print("\nSignature Verification: FAILED")    
    print("The message has been modified.")
